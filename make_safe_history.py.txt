def run_cmd(cmd, cwd, env=None):
    result = subprocess.run(cmd, shell=True, cwd=cwd, env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    return result.returncode == 0

def main():
    source_dir = os.getcwd()
    parent_dir = os.path.dirname(source_dir)
    folder_name = os.path.basename(source_dir)
    
    # Define our output directory next to the current one
    output_dir = os.path.join(parent_dir, f"{folder_name}_git_reconstructed")
    
    if os.path.exists(output_dir):
        print(f"🗑️ Found an old reconstructed folder at {output_dir}. Clearing it...")
        shutil.rmtree(output_dir)
        
    os.makedirs(output_dir, exist_ok=True)
    print(f"📁 Created safe target directory: {output_dir}")

    # Initialize Git workspace inside the output copy directory
    run_cmd("git init", cwd=output_dir)
    run_cmd("git branch -M main", cwd=output_dir)

    # Timeline setup: Start 10 days ago at 09:00 AM
    base_time = datetime.now() - timedelta(days=10)
    base_time = base_time.replace(hour=9, minute=0, second=0, microsecond=0)

    print("⏳ Reconstructing timeline file-by-file inside the isolated copy...")
    
    for day, commits in COMMIT_HISTORY.items():
        time_gap = 420 // len(commits) if len(commits) > 1 else 60
        
        for idx, (commit_msg, target_paths) in enumerate(commits):
            commit_time = base_time + timedelta(days=day-1, minutes=idx * time_gap)
            formatted_time = commit_time.isoformat()
            
            # Move targeted files into the target repository copy
            for path in target_paths:
                source_path = os.path.join(source_dir, path)
                target_path = os.path.join(output_dir, path)
                
                if os.path.exists(source_path):
                    if os.path.isdir(source_path):
                        shutil.copytree(source_path, target_path, dirs_exist_ok=True)
                    else:
                        os.makedirs(os.path.dirname(target_path), exist_ok=True)
                        shutil.copy2(source_path, target_path)
                    
                    # Stage the path inside the target repo
                    run_cmd(f'git add "{path}"', cwd=output_dir)
            
            # Track history to an explicit log file inside the repo to capture text milestones
            history_file = os.path.join(output_dir, "reconstruction_log.md")
            with open(history_file, "a") as f:
                f.write(f"* **{formatted_time}**: {commit_msg}\n")
            run_cmd(f'git add "reconstruction_log.md"', cwd=output_dir)

            # Assign explicit backdates to Git environment variables
            custom_env = os.environ.copy()
            custom_env["GIT_AUTHOR_DATE"] = formatted_time
            custom_env["GIT_COMMITTER_DATE"] = formatted_time

            # Fire commit
            safe_msg = commit_msg.replace("'", "'\"'\"'")
            run_cmd(f"git commit -m '{safe_msg}'", cwd=output_dir, env=custom_env)

    # Bring over absolutely everything left over to ensure no mismatch
    print("📥 Syncing any remaining unmapped background configuration files...")
    shutil.copytree(source_dir, output_dir, dirs_exist_ok=True, ignore=shutil.ignore_patterns("make_safe_history.py", ".git"))
    
    # Catch any remaining drift edits as a final package update
    if run_cmd("git status --porcelain", cwd=output_dir):
        run_cmd("git add .", cwd=output_dir)
        custom_env = os.environ.copy()
        last_date = (base_time + timedelta(days=9, hours=8)).isoformat()
        custom_env["GIT_AUTHOR_DATE"] = last_date
        custom_env["GIT_COMMITTER_DATE"] = last_date
        run_cmd("git commit -m 'chore: minor structural cleanup and final package sync'", cwd=output_dir, env=custom_env)

    print("\n🎉 Success! Your original workspace remains completely safe and untouched.")
    print(f"👉 Navigate to your brand new tracked project folder here:\n   cd {output_dir}")
    print("\n🔗 Run these commands inside that folder to upload to GitHub:")
    print("   git remote add origin <YOUR_GITHUB_REPO_URL>")
    print("   git push -u origin main")

if __name__ == "__main__":
    main()
